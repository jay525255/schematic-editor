cmake_minimum_required(VERSION 3.20)
project(SchematicEditor VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base propgrid xml)
include(${wxWidgets_USE_FILE})
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
  OUTPUT_VARIABLE SE_SOURCE_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
configure_file(src/BuildVersion.h.in generated/BuildVersion.h @ONLY)
add_library(schematic_model STATIC src/ComponentCatalog.cpp src/NetModel.cpp src/SchematicIO.cpp src/SimModel.cpp src/CircuitTools.cpp src/EditingTools.cpp src/KiCadExport.cpp src/KiCadImport.cpp src/DigitalSimulator.cpp)
target_include_directories(schematic_model PUBLIC src "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(schematic_model PUBLIC ${wxWidgets_LIBRARIES})
add_library(schematic_ui STATIC src/MainFrame.cpp src/SchematicCanvas.cpp src/LibraryPanel.cpp
  src/PropertyPanel.cpp src/ComponentEditorDialog.cpp src/SymbolPainter.cpp src/TextReportDialog.cpp)
target_link_libraries(schematic_ui PUBLIC schematic_model)
if(WIN32)
  add_executable(SchematicEditor WIN32 src/minimal.cpp res/SchematicEditor.rc)
else()
  add_executable(SchematicEditor src/minimal.cpp)
endif()
target_link_libraries(SchematicEditor PRIVATE schematic_ui)
add_executable(schematic-cli src/cli.cpp)
target_link_libraries(schematic-cli PRIVATE schematic_model)
if(MSVC)
  add_compile_options(/utf-8)
  foreach(target schematic_model schematic_ui SchematicEditor schematic-cli)
    target_compile_options(${target} PRIVATE /utf-8)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(name core simulation editing_regression)
    add_executable(${name}_tests tests/${name}_tests.cpp)
    target_link_libraries(${name}_tests PRIVATE schematic_model)
    add_test(NAME ${name} COMMAND ${name}_tests "${CMAKE_SOURCE_DIR}/docs/sample-schematic.ses" "${CMAKE_BINARY_DIR}/test-data/${name}")
  endforeach()
  add_executable(kicad_schematic_tests tests/kicad_schematic_tests.cpp)
  target_link_libraries(kicad_schematic_tests PRIVATE schematic_model)
  add_test(NAME kicad_schematic COMMAND kicad_schematic_tests "${CMAKE_SOURCE_DIR}/tests/fixtures/kicad" "${CMAKE_BINARY_DIR}/test-data/kicad-schematic")
  add_executable(gui_native_tests tests/gui_native_tests.cpp)
  target_link_libraries(gui_native_tests PRIVATE schematic_ui)
  add_test(NAME gui_native COMMAND gui_native_tests "${CMAKE_BINARY_DIR}/gui-evidence" "${CMAKE_SOURCE_DIR}/examples/dynamic-and.ses")
  set_tests_properties(gui_native PROPERTIES LABELS gui RUN_SERIAL TRUE TIMEOUT 120)
endif()
install(TARGETS SchematicEditor schematic-cli RUNTIME DESTINATION bin)
